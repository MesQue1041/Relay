### `src/latency.py`

```python
import time
from collections import defaultdict


class LatencyTracker:                     # notebook taht stores latency measurements for each stage of the pipeline
    def __init__(self):
        self.durations = defaultdict(list)

    def record(self, stage: str, seconds: float):
        self.durations[stage].append(seconds)

    def summary(self):
        out = {}
        for stage, values in self.durations.items():    
            out[stage] = {
                "last": values[-1],
                "count": len(values),
                "total": sum(values),
            }
        return out


tracker = LatencyTracker()


class Timer:
    def __init__(self, stage: str):
        self.stage = stage
        self._start = None

    def __enter__(self):
        self._start = time.perf_counter()
        return self

    def __exit__(self, exc_type, exc_val, exc_tb):
        elapsed = time.perf_counter() - self._start
        tracker.record(self.stage, elapsed)
        print(f"[latency] {self.stage}: {elapsed*1000:.0f}ms")
```

### `src/main.py`

```python
from src.audio.capture import MicStream
from src.audio.vad import UtteranceDetector
from src.stt import transcribe
from src.latency import Timer


def main():
    mic = MicStream().start()
    detector = UtteranceDetector()

    print("Listening... speak, then pause for about a second. Ctrl+C to quit.")

    try:
        for frame in mic.frames():
            utterance = detector.push(frame)

            if utterance is None:
                continue 

            with Timer("stt"):
                text = transcribe(utterance)

            if text:
                print(f"You said: {text}")
            else:
                print("(heard something, but couldn't transcribe any words)")

    except KeyboardInterrupt:
        print("\nStopping.")
    finally:
        mic.stop()


if __name__ == "__main__":
    main()
```

### `src/stt.py`

```python
import numpy as np
from faster_whisper import WhisperModel

from src.config import (
    WHISPER_MODEL_SIZE,
    WHISPER_DEVICE,
    WHISPER_COMPUTE_TYPE,
)

_model = WhisperModel(
    WHISPER_MODEL_SIZE,
    device=WHISPER_DEVICE,
    compute_type=WHISPER_COMPUTE_TYPE,
)

def transcribe(pcm_bytes: bytes) -> str:
    """
    pcm_bytes: raw 16-bit PCM audio, mono, at AUDIO_SAMPLE_RATE.
    """
    audio_int16 = np.frombuffer(pcm_bytes, dtype=np.int16)
    audio_float32 = audio_int16.astype(np.float32)

    # This mic's raw signal is consistently very quiet (confirmed via
    # debug logging — peaks around -30 to -40 dBFS even after upstream
    # gain). Rather than guess one fixed gain value that has to be
    # recalibrated per machine/mic, we normalize each utterance so its
    # loudest sample sits near full scale. This makes transcription
    # robust to quiet input instead of relying on absolute level.
    peak = np.abs(audio_float32).max()
    if peak > 0:
        audio_float32 = audio_float32 / peak * 32000.0

    audio_float32 = audio_float32 / 32768.0  # Whisper expects roughly [-1.0, 1.0]

    segments, info = _model.transcribe(audio_float32, language="en")
    return " ".join(segment.text.strip() for segment in segments).strip()
```

### `src/audio/capture.py`

```python
import queue
import numpy as np
import sounddevice as sd

from src.config import AUDIO_SAMPLE_RATE, AUDIO_CHANNELS, VAD_FRAME_DURATION_MS, MIC_GAIN

FRAME_SAMPLES = int(AUDIO_SAMPLE_RATE * VAD_FRAME_DURATION_MS / 1000)


class MicStream:
    def __init__(self):
        self._queue = queue.Queue()
        self._stream = None

    def _callback(self, indata, frames, time_info, status):
        if status:
            print(f"[MicStream] audio status: {status}")

        # Your mic array's raw signal is far below normal speech levels at
        # the driver level (confirmed: peaks of ~10-30 out of a possible
        # 32767, even after raising Windows' recording level). Rather than
        # fight Windows audio enhancements, we just amplify digitally here.
        # int32 avoids overflow during multiplication before we clip back
        # down to int16's valid range.
        amplified = np.clip(
            indata.astype(np.int32) * MIC_GAIN, -32768, 32767
        ).astype(np.int16)
        self._queue.put(amplified.tobytes())

    def start(self):
        self._stream = sd.InputStream(
            samplerate=AUDIO_SAMPLE_RATE,
            channels=AUDIO_CHANNELS,
            dtype="int16",
            blocksize=FRAME_SAMPLES,
            callback=self._callback,
        )
        self._stream.start()
        return self

    def stop(self):
        if self._stream is not None:
            self._stream.stop()
            self._stream.close()
            self._stream = None

    def frames(self):
        while self._stream is not None:
            try:
                yield self._queue.get(timeout=0.5)
            except queue.Empty:
                continue
```

### `src/audio/vad.py`

```python
import webrtcvad

from src.config import (
    AUDIO_SAMPLE_RATE,
    VAD_AGGRESSIVENESS,
    VAD_SILENCE_FRAMES_THRESHOLD,
)


class UtteranceDetector:
    def __init__(self):
        self._vad = webrtcvad.Vad(VAD_AGGRESSIVENESS)
        self._buffer = bytearray()   # accumulated speech audio
        self._silence_run = 0        # consecutive silent frames counted
        self._in_speech = False      # has speech started this utterance?

    def push(self, frame: bytes):
        import numpy as np
        peak = np.abs(np.frombuffer(frame, dtype=np.int16)).max()
        is_speech = self._vad.is_speech(frame, AUDIO_SAMPLE_RATE)
        print(f"peak={peak:5d} is_speech={is_speech}")   # TEMP DEBUG 
        if is_speech:
            self._in_speech = True
            self._silence_run = 0
            self._buffer.extend(frame)
            return None

        # This frame is silence.
        if not self._in_speech:
            return None

        self._silence_run += 1
        self._buffer.extend(frame)  

        if self._silence_run >= VAD_SILENCE_FRAMES_THRESHOLD:
            utterance = bytes(self._buffer)
            self._reset()
            return utterance

        return None

    def _reset(self):
        self._buffer = bytearray()
        self._silence_run = 0
        self._in_speech = False
```
